        function drawRadarChart(isolatedOrgan = 'Spleen') {
            const canvas = document.getElementById('tcmRadarCanvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, 320, 320);

            const centerX = 160;
            const centerY = 160;
            const outerRadius = 110;
            const organs = ['Liver', 'Heart', 'Spleen', 'Lung', 'Kidney'];
            const colors = {

            };

            // Draw concentric pentagon grids
            for (let j = 5; j > 0; j--) {
                const stepRadius = (outerRadius / 5) * j;
                ctx.beginPath();
                for (let i = 0; i < 5; i++) {
                    const angle = (Math.PI * 2 / 5) * i - Math.PI / 2;
                    const x = centerX + Math.cos(angle) * stepRadius;
                    const y = centerY + Math.sin(angle) * stepRadius;
                    if (i === 0) ctx.moveTo(x, y);
                    else ctx.lineTo(x, y);

                ctx.strokeStyle = j === 5 ? 'rgba(135,206,235,0.3)' : 'rgba(135,206,235,0.08)';
                ctx.stroke();
            }

            // Draw element axes
            for (let i = 0; i < 5; i++) {

                ctx.beginPath();
                ctx.moveTo(centerX, centerY);
                ctx.lineTo(centerX + Math.cos(angle) * outerRadius, centerY + Math.sin(angle) * outerRadius);
                ctx.strokeStyle = 'rgba(135,206,235,0.15)';

            }
            ctx.closePath();
            ctx.fillStyle = 'rgba(135,206,235,0.25)';
            ctx.fill();
            ctx.lineWidth = 1.5;
            ctx.strokeStyle = '#87CEEB';
            ctx.stroke();

            // Draw vertices and isolated organ highlighting ring
            coordinates.forEach((pt, i) => {
                ctx.beginPath();
                ctx.arc(pt.x, pt.y, 4, 0, Math.PI * 2);
                ctx.fillStyle = pt.color;
                ctx.fill();

                if (isolatedOrgan === pt.organ) {
                    ctx.beginPath();
                    ctx.arc(pt.x, pt.y, 8, 0, Math.PI * 2);
                    ctx.strokeStyle = pt.color;
                    ctx.lineWidth = 1;
                    ctx.stroke();
                }
            });

            canvas.onclick = function(e) {
                const rect = canvas.getBoundingClientRect();
                const tapX = e.clientX - rect.left;
                const tapY = e.clientY - rect.top;

                // Find closest vertex node
                let closestOrgan = 'Spleen';
                let minDist = 99999;
                
                for (let i = 0; i < 5; i++) {
                    const angle = (Math.PI * 2 / 5) * i - Math.PI / 2;
                    const vX = centerX + Math.cos(angle) * outerRadius;
                    const vY = centerY + Math.sin(angle) * outerRadius;
                    const dist = Math.sqrt((tapX - vX)**2 + (tapY - vY)**2);
                    if (dist < minDist) {
                        minDist = dist;
                        closestOrgan = organs[i];
                    }
                }

                if (minDist < 45) { // Threshold for vertex selection
                    updateSelectedOrgan(closestOrgan);
                    drawRadarChart(closestOrgan);
                }
            };
        }
